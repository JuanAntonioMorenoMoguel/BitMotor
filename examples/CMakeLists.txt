# 10 and 11 need a GLSL compiler for its shaders (glslc or glslangValidator, in the Vulkan SDK).
find_program(BITMOTOR_GLSLC glslc HINTS "$ENV{VULKAN_SDK}/bin" "$ENV{VULKAN_SDK}/Bin")
find_program(BITMOTOR_GLSLANG glslangValidator HINTS "$ENV{VULKAN_SDK}/bin" "$ENV{VULKAN_SDK}/Bin")
if(BITMOTOR_GLSLC OR BITMOTOR_GLSLANG)
    BitMotor_Example(10_CustomRenderer)
    BitMotor_Shaders(BitMotor_10_CustomRenderer 10_CustomRenderer/Shaders)
    BitMotor_Example(11_Compute)
    BitMotor_Shaders(BitMotor_11_Compute 11_Compute/Shaders)
else()
    message(STATUS "BitMotor: 10_CustomRenderer and 11_Compute skipped (no glslc or glslangValidator: install the Vulkan SDK)")
endif()
